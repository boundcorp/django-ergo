"""Canonical Article path helpers."""

from __future__ import annotations

import posixpath

def legacy_hierarchy_to_relative_path(hierarchy_code: str) -> str:
    """Convert a legacy hierarchy code into slash-separated path segments.

    Legacy codes such as ``011`` become ``0/1/1``. Values that already use
    slash-delimited hierarchy tokens are preserved, so ``0/1/1/10`` keeps the
    multi-character ``10`` segment intact.
    """
    value = hierarchy_code.strip().strip("/")
    if not value:
        raise ValueError("A legacy hierarchy code cannot be empty.")
    if "/" in value:
        segments = value.split("/")
    else:
        segments = list(value)
    if any(not segment or segment in {".", ".."} for segment in segments):
        raise ValueError(f"Invalid legacy hierarchy path: {hierarchy_code!r}")
    return "/".join(segments)


def validate_relative_path(relative_path: str) -> str:
    """Validate and normalize a repository-relative POSIX path."""
    value = relative_path.strip().replace("\\", "/")
    normalized = posixpath.normpath(value)
    if not value or normalized in {".", ".."} or normalized.startswith("../"):
        raise ValueError(f"Invalid relative path: {relative_path!r}")
    if normalized.startswith("/"):
        raise ValueError(f"Relative path must not be absolute: {relative_path!r}")
    return normalized
